        "gamma = 0.9\n",

        "\n",
        "# Deterministic policy\n",
        "policy = {\n",
        "    \"S0\": \"Down\",\n",
        "    \"S1\": \"Right\"\n",
        "}\n",
        "\n",
        "# Transition model: (next state, probability, reward)\n",
        "transitions = {\n",
        "    (\"S0\", \"Down\"): [\n",
        "        (\"Wall\", 0.1, -1),\n",
        "        (\"S1\", 0.9, -1)\n",
        "    ],\n",
        "    (\"S0\", \"Right\"): [\n",
        "        (\"Wall\", 1.0, -1)\n",
        "    ],\n",
        "    (\"S1\", \"Right\"): [\n",
        "        (\"G\", 0.9, 100),\n",
        "        (\"S0\", 0.1, -1)\n",
        "    ],\n",
        "    (\"S1\", \"Up\"): [\n",
        "        (\"S0\", 1.0, -1)\n",
        "    ]\n",
        "}\n",
        "\n",
        "# Initialize state values\n",
        "V = {\n",
        "    \"S0\": 0.0,\n",
        "    \"S1\": 0.0\n",
        "}\n",
        "\n",
        "iterations = 0\n",
        "\n",
        "# Iterative policy evaluation\n",
        "while True:\n",
        "    old = V.copy()\n",
        "\n",
        "    new_V = {}\n",
        "\n",
        "    for state in [\"S0\", \"S1\"]:\n",
        "        action = policy[state]\n",
        "        value = 0.0\n",
        "\n",
        "        for next_state, prob, reward in transitions[(state, action)]:\n",
        "            value += prob * (\n",
        "                reward + gamma * old.get(next_state, 0.0)\n",
        "            )\n",
        "\n",
        "        new_V[state] = value\n",
        "\n",
        "    delta = max(\n",
        "        abs(new_V[\"S0\"] - old[\"S0\"]),\n",
        "        abs(new_V[\"S1\"] - old[\"S1\"])\n",
        "    )\n",
        "\n",
        "    V = new_V\n",
        "    iterations += 1\n",
        "\n",
        "    if delta < theta:\n",
        "        break\n",
        "\n",
        "# Calculate action values after convergence\n",
